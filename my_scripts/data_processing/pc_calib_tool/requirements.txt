# PC (native desktop) version of the iPad<->Mocap calibration tool.
# Recommended clean setup (see README):
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements.txt
#
# PySide6  = the modern, maintained Qt6 binding for the desktop UI.
# opencv-python-headless = OpenCV WITHOUT its bundled Qt GUI.  Using the
#   regular opencv-python here breaks Qt (it sets QT_QPA_PLATFORM_PLUGIN_PATH
#   to its own cv2/qt/plugins, so PySide6 can't load the xcb platform plugin
#   and the app aborts).  Headless has everything this tool uses.
PySide6>=6.6
opencv-python-headless>=4.8
numpy
scipy
matplotlib
ezc3d>=1.5
